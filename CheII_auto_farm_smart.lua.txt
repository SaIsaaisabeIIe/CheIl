local function safeFire(container, remoteName, ...)
    if not container then
        warn("[ETW] Container remote tidak tersedia: " .. tostring(remoteName))
        return false
    end
    local remote = container:FindFirstChild(remoteName)
    if not remote then
        warn("[ETW] Remote tidak ditemukan: " .. tostring(remoteName))
        return false
    end
    local ok, err = pcall(function(...)
        remote:FireServer(...)
    end, ...)
    if not ok then
        warn("[ETW] Remote gagal dipanggil: " .. tostring(err))
        return false
    end
    return true
end

-- ===================== [ID] JADWAL TELEPORT MEGA =====================
-- Mega tidak lagi dikirim bersamaan dengan Normal. Mega dikirim
-- MEGA_TELEPORT_DELAY detik setelah Auto Farm berjalan, dan sisa waktunya
-- (remaining) ditampilkan di label panel serta overlay Auto Farm.
local ticketTimerLabel -- [ID] dibuat di bawah; dipakai juga oleh Auto Farm
local megaCountdown = nil -- nil = tidak ada hitung mundur, angka = sisa detik
local megaTeleportSent = false
local megaTeleportStatus = "WAITING" -- WAITING / SENT / ERROR / DONE

local function fireTeleportOnce(kind)
    if not TeleportEvent then
        warn("[CHEII] RequestTeleport tidak ditemukan; teleport " .. tostring(kind) .. " dilewati.")
        return false, "TELEPORT REMOTE MISSING"
    end
    local ok, err = pcall(function()
        TeleportEvent:FireServer(kind)
    end)
    if not ok then
        return false, err
    end
    return true
end

local function startMegaTeleportCountdown()
    if megaTeleportSent then
        return
    end
    task.spawn(function()
        for remaining = MEGA_TELEPORT_DELAY, 1, -1 do
            megaCountdown = remaining
            if ticketTimerLabel then
                ticketTimerLabel.Text = "MEGA: " .. remaining .. "s"
            end
            task.wait(1)
        end

        megaCountdown = 0
        local ok, err = fireTeleportOnce("Mega")
        megaCountdown = nil

        if ok then
            megaTeleportSent = true
            megaTeleportStatus = "SENT"
            if ticketTimerLabel then
                ticketTimerLabel.Text = "MEGA: SENT"
            end
            -- Status SENT hanya sebentar, setelah itu kembali ke TICKET biasa.
            task.delay(5, function()
                if megaTeleportStatus == "SENT" then
                    megaTeleportStatus = "DONE"
                end
            end)
        else
            megaTeleportStatus = "ERROR"
            if ticketTimerLabel then
                ticketTimerLabel.Text = "MEGA: ERROR"
            end
            warn("[CHEII] Teleport Mega:", err)
        end
    end)
end

-- 窗口拖拽
function UI:MakeDraggable()
    local dragging = false
    local dragStart = Vector2.new()
    local startPos = UDim2.new()
    self.TitleBar.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = self.Window.Position
            UserInputService.MouseIconEnabled = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            self.Window.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
            UserInputService.MouseIconEnabled = true
        end
    end)
end
-- 收起/展开窗口
function UI:ToggleCollapse()
    self.IsCollapsed = not self.IsCollapsed
    if self.IsCollapsed then
        -- 收起
        -- 同步收起所有展开的子菜单
        for _, section in ipairs(self.Sections) do
            if section.IsExpanded then
                section.IsExpanded = false
                section.ToggleBtn.Text = "+"
                TweenService:Create(section.Content, TweenInfo.new(0.2), {
                    Size = UDim2.new(1, 0, 0, 0)
                }):Play()
                -- 更新主容器高度
                self.CurrentY = self.CurrentY - section.ContentY
            end
        end
        TweenService:Create(self.Window, TweenInfo.new(0.2), {
            Size = UDim2.new(0, 150, 0, 40)
        }):Play()
        self.CollapseBtn.Text = "+"
        self.ContentContainer.Visible = false
    else
        -- 展开
        -- 重新计算CurrentY和所有子菜单的位置
        self.CurrentY = 0
        local currentOffset = 0
        for _, section in ipairs(self.Sections) do
            -- 设置子菜单的位置
            section.Frame.Position = UDim2.new(0, 0, 0, currentOffset)
            section.Content.Position = UDim2.new(0, 0, 0, currentOffset + 50)
            -- 更新当前偏移量和CurrentY
            currentOffset = currentOffset + 50
            self.CurrentY = self.CurrentY + 50
        end
        -- 计算正确的窗口高度，与子菜单展开/收起时的逻辑一致
        local newHeight = math.max(150, 40 + self.CurrentY + 20) -- 40 是标题栏高度，20 是底部边距
        TweenService:Create(self.Window, TweenInfo.new(0.2), {
            Size = UDim2.new(0, 150, 0, newHeight)
        }):Play()
        self.CollapseBtn.Text = "-"
        -- 确保MenuContainer大小正确
        self.MenuContainer.Size = UDim2.new(1, 0, 1, 0)
        -- 恢复ContentContainer的相对大小
        self.ContentContainer.Size = UDim2.new(1, 0, 1, -40)
        -- 确保所有子菜单都可见
        for _, section in ipairs(self.Sections) do
            section.Frame.Visible = true
            section.Content.Visible = true
        end
        task.wait(0.2)
        self.ContentContainer.Visible = true
    end
end

            events = char:WaitForChild("Events", 10)
            if not events then
                warn("[ETW] Autofarm dinonaktifkan: remote 'Events' tidak ada di "
                    .. "karakter (kemungkinan nama remote berubah setelah update game).")
                return
            end
            eat = events:WaitForChild("Eat", 10)
            grab = events:WaitForChild("Grab", 10)
            sell = events:WaitForChild("Sell", 10)
            chunk = char:WaitForChild("CurrentChunk", 10)
            sendTrack = char:WaitForChild("SendTrack", 10)
            radius = char:WaitForChild("Radius", 10)
            if not (size and eat and grab and sell and chunk and sendTrack and radius) then
                warn("[ETW] Autofarm dinonaktifkan: object game tidak lengkap "
                    .. "(Size/CurrentChunk/SendTrack/Radius atau Eat/Grab/Sell).")
                return
            end
            autoConn = game["Run Service"].Heartbeat:Connect(function(dt)
                if not autofarm then
                    autoConn:Disconnect()
                    return
                end
                local ran = tick() - startTime
                local hours = math.floor(ran / 60 / 60)
                local minutes = math.floor(ran / 60)
                local seconds = math.floor(ran)
                local eatMinutes = math.floor(eatTime / 60)
                local eatSeconds = math.floor(eatTime)
                local y = bedrock.Position.Y + bedrock.Size.Y / 2 + hum.HipHeight + root.Size.Y / 2
                local sizeAdd = LocalPlayer.Upgrades.Multiplier.Value / 100
                local addAmount = LocalPlayer.Upgrades.MaxSize.Value / sizeAdd
                local sellTime = addAmount / 2
                local sellMinutes = math.floor(sellTime / 60)
                local sellSeconds = math.floor(sellTime)
                local secondEarn = math.floor(((LocalPlayer.Upgrades.MaxSize.Value + 0.5) ^ 2 - 0.25) / 2 * 100 / sellTime)
                local minuteEarn = secondEarn * 60
                local hourEarn = minuteEarn * 60
                local dayEarn = hourEarn * 24
                -- [ID] Sisa waktu sebelum RequestTeleport("Mega") dikirim.
                local megaLine = ""
                if megaCountdown and megaCountdown > 0 then
                    megaLine = "\nMega Teleport: " .. megaCountdown .. "s"
                elseif megaTeleportStatus == "SENT" then
                    megaLine = "\nMega Teleport: SENT"
                end
                text.Text = ""
                    .. "\nRun Time: " .. string.format("%ih%im%is", hours, minutes % 60, seconds % 60)
                    .. "\nActual Time: " .. string.format("%im%is", eatMinutes % 60, eatSeconds % 60)
                    .. "\nEstimated Time: " .. string.format("%im%is", sellMinutes % 60, sellSeconds % 60)
                    .. "\nDaily Income: " .. dayEarn
                    .. "\nBlock Count: " .. numChunks
                    .. megaLine
                hum:ChangeState(Enum.HumanoidStateType.Physics)
                grab:FireServer()
                root.Anchored = false
                eat:FireServer()
                sendTrack:FireServer()
                if chunk.Value then
                    if timer > 0 then
                        numChunks += 1
                    end
                    timer = 0
                    grabTimer += dt
                else
                    timer += dt
                    grabTimer = 0
                end
                if timer > 60 then
                    hum.Health = 0
                    timer = 0
                    numChunks = 0
                end
                if grabTimer > 15 then
                    size.Value = LocalPlayer.Upgrades.MaxSize.Value
                end
                if (size.Value >= LocalPlayer.Upgrades.MaxSize.Value)
                    or timer > 8
                then
                    if timer < 8 then
                        sell:FireServer()
                        if not sellDebounce then
                            local args = {
                                {
                                    MapTime = -1,
                                    Paused = true
                                }
                            }
                            safeFire(Events, "SetServerSettings", unpack(args))
                        end
                        sellDebounce = true
                    else
                        local args = {
                            {
                                MapTime = -1,
                                Paused = true
                            }
                        }
                        safeFire(Events, "SetServerSettings", unpack(args))
                    end
                    numChunks = 0
                elseif size.Value == 0 then
                    if sellDebounce then
                        local currentEatTime = tick()
                        eatTime = currentEatTime - lastEatTime
                        lastEatTime = currentEatTime
                        sellCount += 1
                    end
                    sellDebounce = false
                end
                if farmMoving then
                    local bound = 300
                    local startPos = CFrame.new(-bound/2, y, -bound/2)
                    local r = radius.Value * 1.1
                    local dist = (r * numChunks)
                    local x = dist % bound
                    local z = math.floor(dist / bound) * r
                    local offset = CFrame.new(x, 0, z + r * 2)
                    if z > bound then
                        local args = {
                            {
                                MapTime = -1,
                                Paused = true
                            }
                        }
                        safeFire(Events, "SetServerSettings", unpack(args))
                        numChunks = 0
                    end
                    root.CFrame = startPos * offset
                else
                    root.CFrame = CFrame.new(0, y, 0)
                end
            end)
            hum.Died:Connect(function()
                autoConn:Disconnect()
                local args = {
                    {
                        MapTime = -1,
                        Paused = true
                    }
                }
                safeFire(Events, "SetServerSettings", unpack(args))
            end)
            -- [ID] FindFirstChild + pesan, bukan WaitForChild tanpa batas waktu.
            local etwManager = char:FindFirstChild("LocalChunkManager")
            if etwManager then
                etwManager.Enabled = false
            else
                warn("[ETW] 'LocalChunkManager' tidak ditemukan, bagian ini dilewati.")
            end
            local etwAnimate = char:FindFirstChild("Animate")
            if etwAnimate then
                etwAnimate.Enabled = false
            end
        end
        if LocalPlayer.Character then
            task.spawn(onCharAdd, LocalPlayer.Character)
        else
            task.spawn(onCharAdd, LocalPlayer.CharacterAdded:Wait())
        end
        local charAddConn = LocalPlayer.CharacterAdded:Connect(onCharAdd)
        while autofarm do
            local dt = task.wait()
            if workspace:FindFirstChild("Loading") then
                workspace.Loading:Destroy()
            end
            if map and chunks then
                if showMap then
                    map.Parent, chunks.Parent = workspace, workspace
                else
                    map.Parent, chunks.Parent = nil, nil
                end
            end
        end
        charAddConn:Disconnect()
        if autoConn then
            autoConn:Disconnect()
        end
        if map and chunks then
            map.Parent, chunks.Parent = workspace, workspace
        end
        if hum then
            hum:ChangeState(Enum.HumanoidStateType.GettingUp)
        end
        bedrock:Destroy()
        local etwRestoreChar = LocalPlayer.Character
        local etwRestoreManager = etwRestoreChar and etwRestoreChar:FindFirstChild("LocalChunkManager")
        if etwRestoreManager then
            etwRestoreManager.Enabled = true
        end
        local etwRestoreAnimate = etwRestoreChar and etwRestoreChar:FindFirstChild("Animate")
        if etwRestoreAnimate then
            etwRestoreAnimate.Enabled = true
        end
        text:Destroy()
        autoFarmRunning = false
    end)()
end

local function formatTicketTime(seconds)
    local hours = math.floor(seconds / 3600)
    local minutes = math.floor((seconds % 3600) / 60)
    local secs = seconds % 60
    return string.format("%02d:%02d:%02d", hours, minutes, secs)
end

local function claimSmallRewardsOnce()
    if not RewardEvent then
        warn("[CHEII] RewardEvent tidak ditemukan; Small Reward dilewati.")
        return
    end
    if not TimedRewards then
        warn("[CHEII] TimedRewards tidak ditemukan; reward dilewati.")
        return
    end

    -- SmallReward, lalu GetChildren() index 2 sampai 9.
    local smallReward = TimedRewards:FindFirstChild("SmallReward")
    if smallReward then
        local ok, err = pcall(function()
            RewardEvent:FireServer(smallReward)
        end)
        if not ok then warn("[CHEII] SmallReward gagal:", err) end
    else
        warn("[CHEII] SmallReward tidak ditemukan.")
    end

    local children = TimedRewards:GetChildren()
    for index = 2, 9 do
        local reward = children[index]
        if reward then
            local ok, err = pcall(function()
                RewardEvent:FireServer(reward)
            end)
            if not ok then warn("[CHEII] Reward index " .. index .. " gagal:", err) end
        else
            warn("[CHEII] TimedRewards:GetChildren()[" .. index .. "] tidak ada.")
        end
    end
end

local function spinOnce()
    if not SpinEvent then
        warn("[CHEII] SpinEvent tidak ditemukan; Spin dilewati.")
        return
    end
    local ok, err = pcall(function()
        SpinEvent:FireServer()
    end)
    if not ok then warn("[CHEII] SpinEvent gagal:", err) end
end

local function fireTeleportNormalOnce()
    if not TeleportEvent then
        warn("[CHEII] RequestTeleport tidak ditemukan; teleport dilewati.")
        ticketTimerLabel.Text = "TELEPORT REMOTE MISSING"
        return
    end

    -- Normal dipanggil tepat satu kali saat sisa waktu 00:00 (seperti sebelumnya).
    local normalOk, normalErr = fireTeleportOnce("Normal")
    if normalOk then
        ticketTimerLabel.Text = "TELEPORT SENT"
    else
        warn("[CHEII] Teleport Normal:", normalErr)
        ticketTimerLabel.Text = "TELEPORT ERROR"
    end

    -- [ID] Mega sudah dijadwalkan 5 detik setelah Auto Farm berjalan. Kalau
    -- Auto Farm tidak pernah jalan (OFF ditekan sebelum 10 detik), Mega
    -- dikirim di sini supaya fitur teleport Mega tidak hilang.
    if not megaTeleportSent then
        local megaOk, megaErr = fireTeleportOnce("Mega")
        if megaOk then
            megaTeleportSent = true
            megaTeleportStatus = "SENT"
        else
            warn("[CHEII] Teleport Mega:", megaErr)
        end
    end
end

autoSection:CreateButton("OFF", function()
    -- OFF hanya mematikan Auto Farm. Countdown dan jadwal reward/spin/teleport
    -- tetap berjalan.
    autoFarmCancelled = true
    autofarm = false
    config.autofarm = false
    autoSaveConfig()
end)

-- Countdown berulang setiap 01:33:00.
-- Auto Farm hanya dicoba sekali, 10 detik setelah script dijalankan.
task.spawn(function()
    local elapsedTotal = 0
    local autoFarmStartAttempted = false

    while true do
        for remaining = TICKET_TIMER, 0, -1 do
            -- [ID] Selama hitung mundur Mega berjalan, label dipegang oleh
            -- startMegaTeleportCountdown supaya teksnya tidak saling menimpa.
            if not megaCountdown then
                ticketTimerLabel.Text = "TICKET: " .. formatTicketTime(remaining)
            end

            if not autoFarmStartAttempted
                and elapsedTotal == INITIAL_DELAY
                and not autoFarmCancelled then
                autoFarmStartAttempted = true
                startAutoFarm()
            end

            if remaining == 10 then
                ticketTimerLabel.Text = "TICKET: REWARDS"
                claimSmallRewardsOnce()
            elseif remaining == 8 then
                ticketTimerLabel.Text = "TICKET: SPIN"
                spinOnce()
            elseif remaining == 0 then
                ticketTimerLabel.Text = "TICKET: TELEPORT"
                fireTeleportNormalOnce()
            end

            if remaining > 0 then
                task.wait(1)
                elapsedTotal += 1
            end
        end

        -- Siklus tiket baru dimulai setelah teleport pada 00:00.
        task.wait(1)
        elapsedTotal += 1
    end
end)
